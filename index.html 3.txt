        // ===== PROGRESS MERGE LOGIC SECTION (pure helpers that decide how two copies of progress combine) =====
        // <progress-core>
        // Pure helpers (no DOM / Firebase) that decide how two copies of progress combine.
        const Progress = (() => {
            const isObj = v => v !== null && typeof v === 'object' && !Array.isArray(v);
            const ms = s => { const t = Date.parse(s); return Number.isNaN(t) ? 0 : t; };
            const hasText = v => typeof v === 'string' && v !== '';
            const stamp = h => isObj(h) ? Math.max(ms(h.date), ms(h.confAt)) : 0;

            // history:     { questionId: { correct, date, conf, confAt? } }
            // bookmarks:   [questionId]                       (what the UI reads)
            // bookmarkLog: { questionId: { on, t } }          (latest toggle, so un-bookmarks can sync)
            // notes:       { questionId: text }               (what the UI reads)
            // noteTimes:   { questionId: ISO time }           (latest edit, so deleted notes can sync)
            // resetAt:     ISO time of the last "Reset Progress" made while signed in
            const empty = () => ({ history: {}, bookmarks: [], notes: {}, bookmarkLog: {}, noteTimes: {}, resetAt: null });

            function normalize(p) {
                p = isObj(p) ? p : {};
                return {
                    history: isObj(p.history) ? p.history : {},
                    bookmarks: Array.isArray(p.bookmarks) ? p.bookmarks : [],
                    notes: isObj(p.notes) ? p.notes : {},
                    bookmarkLog: isObj(p.bookmarkLog) ? p.bookmarkLog : {},
                    noteTimes: isObj(p.noteTimes) ? p.noteTimes : {},
                    resetAt: typeof p.resetAt === 'string' ? p.resetAt : null
                };
            }

            function hasData(p) {
                p = normalize(p);
                return Object.keys(p.history).length > 0 || p.bookmarks.length > 0 || Object.keys(p.notes).length > 0 ||
                       Object.keys(p.bookmarkLog).length > 0 || Object.keys(p.noteTimes).length > 0;
            }

            // Combine two copies. Order only matters for exact ties, where `b` wins.
            function merge(a, b) {
                a = normalize(a); b = normalize(b);
                const out = empty();
                const resetMs = Math.max(ms(a.resetAt), ms(b.resetAt));
                out.resetAt = ms(b.resetAt) > ms(a.resetAt) ? b.resetAt : a.resetAt;

                // Answers: the most recent attempt at each question wins.
                for (const id of new Set([...Object.keys(a.history), ...Object.keys(b.history)])) {
                    const ha = a.history[id], hb = b.history[id];
                    const best = !isObj(ha) ? hb : !isObj(hb) ? ha : (stamp(ha) > stamp(hb) ? ha : hb);
                    if (isObj(best) && stamp(best) >= resetMs) out.history[id] = best;
                }

                // Bookmarks: the latest on/off toggle wins; a bookmark with no log entry counts as "on, long ago".
                const setA = new Set(a.bookmarks.map(String)), setB = new Set(b.bookmarks.map(String));
                const original = new Map();   // keeps ids as numbers/strings exactly as the question bank has them
                for (const v of [...a.bookmarks, ...b.bookmarks]) if (!original.has(String(v))) original.set(String(v), v);
                const logOf = (p, set, k) => isObj(p.bookmarkLog[k]) ? p.bookmarkLog[k] : (set.has(k) ? { on: true, t: null } : null);
                for (const k of new Set([...original.keys(), ...Object.keys(a.bookmarkLog), ...Object.keys(b.bookmarkLog)])) {
                    const ea = logOf(a, setA, k), eb = logOf(b, setB, k);
                    let best;
                    if (!ea || !eb) best = ea || eb;
                    else if (ms(ea.t) !== ms(eb.t)) best = ms(ea.t) > ms(eb.t) ? ea : eb;
                    else best = ea.on ? ea : eb;                 // tie: keep the bookmark
                    if (ms(best.t) < resetMs) continue;
                    out.bookmarkLog[k] = { on: !!best.on, t: best.t || null };
                    if (best.on) out.bookmarks.push(original.has(k) ? original.get(k) : k);
                }

                // Notes: the latest edit wins (an emptied note is remembered so the deletion syncs too).
                for (const k of new Set([...Object.keys(a.notes), ...Object.keys(b.notes), ...Object.keys(a.noteTimes), ...Object.keys(b.noteTimes)])) {
                    const ta = ms(a.noteTimes[k]), tb = ms(b.noteTimes[k]);
                    const useA = ta !== tb ? ta > tb : (!hasText(b.notes[k]) && hasText(a.notes[k]));
                    const src = useA ? a : b;
                    if (Math.max(ta, tb) < resetMs) continue;
                    if (hasText(src.notes[k])) out.notes[k] = src.notes[k];
                    if (src.noteTimes[k]) out.noteTimes[k] = src.noteTimes[k];
                }
                return out;
            }

            return { empty, normalize, hasData, merge };
        })();
        // ===== END OF PROGRESS MERGE LOGIC SECTION =====

        // ===== LOCAL DEVICE STORAGE SECTION (read, write and remove progress in localStorage) =====
        // This device's copy of the progress (guest copy, or one copy per signed-in account).
        const Local = {
            key: (uid, bankId) => uid ? `taumd_user_${uid}_progress_${bankId}` : `taumd_progress_${bankId}`,
            read(uid, bankId) {
                try {
                    const raw = localStorage.getItem(this.key(uid, bankId));
                    return Progress.normalize(raw ? JSON.parse(raw) : null);
                } catch (e) { return Progress.empty(); }
            },
            write(uid, bankId, progress) {
                try { localStorage.setItem(this.key(uid, bankId), JSON.stringify(progress)); return true; }
                catch (e) { console.warn('Could not save progress on this device:', e); return false; }
            },
            remove(uid, bankId) {
                try { localStorage.removeItem(this.key(uid, bankId)); } catch (e) { /* ignore */ }
            }
        };
        // </progress-core>
        // ===== END OF LOCAL DEVICE STORAGE SECTION =====

                async syncBank(uid, bankId) {
                    if (this.status !== 'ready' || this.uid() !== uid) return false;
                    setSyncStatus('syncing');
                    // The bank on screen is the freshest copy; any other bank is read back from this device.
                    const live = () => state.bankId === bankId ? state.progress : Local.read(uid, bankId);
                    try {
                        const { doc, runTransaction } = this.F;
                        const ref = doc(this.db, 'users', uid, 'banks', bankId);
                        const merged = await runTransaction(this.db, async tx => {
                            if (this.uid() !== uid) throw new Error('identity-changed');   // never write one account's data into another's
                            const snap = await tx.get(ref);
                            const m = Progress.merge(live(), snap.exists() ? snap.data() : null);
                            tx.set(ref, JSON.parse(JSON.stringify(m)));
                            return m;
                        });
                        if (this.uid() !== uid) return false;
                        const latest = Progress.merge(live(), merged);      // keep anything done while the sync was running
                        Local.write(uid, bankId, latest);
                        if (state.bankId === bankId) { state.progress = latest; refreshView(); }
                        setSyncStatus('synced');
                        return true;
                    } catch (err) {
                        if (err && err.message === 'identity-changed') return false;
                        console.error('Cloud sync failed:', err);
                        const code = (err && err.code) || '';
                        if (code === 'permission-denied') console.error('Firestore rules are blocking access. See the setup notes at the top of this script.');
                        const offline = !navigator.onLine || /unavailable|network|offline/i.test(`${code} ${(err && err.message) || ''}`);
                        setSyncStatus(offline ? 'offline' : 'error', offline ? '' : (code || 'unknown error'));
                        return false;

                // ----- LOAD PROGRESS (this device and cloud) -----
                async loadBankProgress() {
                    if(!state.bankId) return;
                    const bankId = state.bankId;
                    const uid = account.uid();

                    // 1. Instant local load: the account's copy on this device if signed in, otherwise the guest copy
                    let progress = Local.read(uid, bankId);

                    // 2. Signed in: fold in anything saved here as a guest, then retire the guest copy
                    if (uid) {

                    const uid = account.uid();
                    Local.write(uid, state.bankId, state.progress);
                    if (uid) account.schedulePush(uid, state.bankId);
                },

                resetLocalObj() {
                    state.progress = Progress.empty();
                },

                toggleBookmark(qId) {
                    const p = state.progress;
                    const idx = p.bookmarks.indexOf(qId);
                    if(idx > -1) p.bookmarks.splice(idx, 1);
                    else p.bookmarks.push(qId);
                    p.bookmarkLog[qId] = { on: idx === -1, t: new Date().toISOString() };
                    this.saveBankProgress();
                    return idx === -1;
                },

                saveNote(qId, text) {
                    if(!text.trim()) delete state.progress.notes[qId];
                    else state.progress.notes[qId] = text;
                    state.progress.noteTimes[qId] = new Date().toISOString();
                    this.saveBankProgress();
                },

                saveAnswer(qId, isCorrect, confLevel = null) {
                    state.progress.history[qId] = { correct: isCorrect, date: new Date().toISOString(), conf: confLevel };
                    this.saveBankProgress();
                },

                // ----- RESET AND EXPORT -----
                resetCurrentBank() {
                    const where = account.user ? "on this device and in your account" : "on this device";
                    ui.showConfirm("Reset Progress", `Are you sure? This will delete all history, bookmarks, and notes for this question bank ${where}.`, async () => {
                        const uid = account.uid();
                        this.resetLocalObj();
                        // A signed-in reset is time-stamped so your other devices drop their older progress when they sync
                        if (uid) state.progress.resetAt = new Date().toISOString();
                        this.saveBankProgress();
                        if (uid) await account.flush();

                        if (uid && account.syncState !== 'synced') ui.showAlert("Progress reset", "Progress was reset on this device. It will be cleared from your account the next time it syncs.");
                        else ui.showAlert("Success", "Progress has been reset.");
                        if(state.session) ui.navigate('dashboard');
                        else dashboard.render();
                    });
                },

                exportData() {
                    if(!state.bankId) return ui.showAlert("Error", "No bank loaded.");
                    const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(state.progress, null, 2));
                    const dl = document.createElement('a');
                    dl.setAttribute("href", dataStr);
                    dl.setAttribute("download", `taumd_export_${state.bankId}.json`);
                    document.body.appendChild(dl);
                    dl.click();
                    dl.remove();
                }
            };
            // ===== END OF STORAGE SECTION =====

            // ===== DATA LOADING SECTION (load question banks, JSON upload, text formatting) =====
            const data = {
                // ----- LOAD QUESTION BANK (manifest, questions.json, manual upload) -----
                async initLoad() {
                    ui.navigate('loading');
                    const loadingTextEl = $('loading-text');
                    
                    try {
                        if (loadingTextEl) loadingTextEl.textContent = "Checking for JSON manifest...";
                        
                        const manifestRes = await fetch('./manifest.json').catch(() => null);
                        if (manifestRes && manifestRes.ok) {
                            const files = await manifestRes.json();
                            if (Array.isArray(files) && files.length > 0) {
                                if (loadingTextEl) loadingTextEl.textContent = `Loading ${files.length} question banks...`;
                                
                                let allQuestions = [];
                                let baseMetadata = { 
                                    title: "Combined TAUMD Bank", 
                                    description: "Multiple JSON files loaded via manifest." 
                                };
                                
                                const fetchPromises = files.map(file => fetch(file).then(r => r.ok ? r.json() : null).catch(() => null));
                                const results = await Promise.all(fetchPromises);
                                
                                results.forEach((json, index) => {
                                    if (json && json.questions) {
                                        allQuestions = allQuestions.concat(json.questions);
                                        if (index === 0 && json.metadata) baseMetadata = json.metadata; 
                                    }
                                });
                                
                                if(allQuestions.length > 0) {
                                    await this.process({ metadata: baseMetadata, questions: allQuestions });
                                    return;
                                }
                            }
                        }
                        
                        if (loadingTextEl) loadingTextEl.textContent = "Loading default question bank...";
                        const response = await fetch('./questions.json');
                        if (!response.ok) throw new Error("HTTP error " + response.status);
                        const json = await response.json();
                        await this.process(json);

                    } catch (err) {
                        console.warn("Could not fetch local files. Awaiting manual upload.", err);
                        if (loadingTextEl) loadingTextEl.textContent = "Ready for manual upload.";
                        ui.navigate('dashboard'); 
                    }
                },
                handleFileUpload(e) {
                    const file = e.target.files[0];
                    if(!file) return;
                    ui.navigate('loading');
                    const reader = new FileReader();
                    reader.onload = async (ev) => {
                        try {
                            const json = JSON.parse(ev.target.result);
                            await data.process(json);
                        } catch(err) {
                            ui.showAlert("Invalid JSON", "The selected file is not a valid JSON structure.");
                            ui.navigate('dashboard');
                        }
                    };
                    reader.readAsText(file);
                },
                // ----- PROCESS BANK AND FORMAT TEXT -----
                async process(json) {
                    if (!json.questions || !Array.isArray(json.questions)) {
                        ui.showAlert("Error", "Invalid bank schema. Missing 'questions' array.");
                        ui.navigate('dashboard');
                        return;
                    }
                    
                    state.bank = json;
                    const title = json.metadata?.title || "Unknown Bank";
                    state.bankId = title.toLowerCase().replace(/[^a-z0-9]/g, '_');
                    
                    $('nav-bank-title').textContent = title;
                    $('dash-bank-desc').textContent = json.metadata?.description || "Select a mode to begin.";
                    
                    // Await the cloud fetch before showing dashboard
                    await storage.loadBankProgress();
                    
                    dashboard.render();
                    ui.navigate('dashboard');
                },
                escapeHTML(str) {
                    if(typeof str !== 'string') return '';
                    return str.replace(/[&<>'"]/g, 
                        tag => ({
                            '&': '&amp;',
                            '<': '&lt;',
                            '>': '&gt;',
                            "'": '&#39;',
                            '"': '&quot;'
                        }[tag] || tag)
                    );
                },
                formatContent(text) {
                    if(!text) return "";
                    let html = this.escapeHTML(text);
                    html = html.replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>');
                    html = html.replace(/\n/g, '<br>');
                    return html;
                }
            };
            // ===== END OF DATA LOADING SECTION =====

            // ===== DASHBOARD SECTION (stats cards and resume panel) =====
            const dashboard = {
                render() {
                    if(!state.bank) return;
                    
                    const total = state.bank.questions.length;
                    const hist = Object.values(state.progress.history);
                    const attempted = hist.length;
                    const correct = hist.filter(h => h.correct).length;
                    const acc = attempted > 0 ? Math.round((correct / attempted) * 100) : 0;
                    const bookmarks = state.progress.bookmarks.length;

                    $('stat-total').textContent = total;
                    $('stat-attempted').textContent = attempted;
                    $('stat-accuracy').textContent = acc + '%';
                    $('stat-bookmarks').textContent = bookmarks;

                    const mistakes = hist.length - correct;
                    const btnMistakes = $('btn-review-mistakes');
                    if (mistakes > 0) {
                        btnMistakes.classList.remove('hidden');
                        btnMistakes.textContent = `Mistakes (${mistakes})`;
                    } else {
                        btnMistakes.classList.add('hidden');
                    }

                    if (state.session && !state.session.isFinished) {
                        $('resume-panel').classList.remove('hidden');
                        $('resume-text').textContent = `${state.session.mode === 'exam' ? 'Exam' : 'Tutor'} block in progress (${state.session.currentIndex + 1} / ${state.session.questions.length})`;
                    } else {
                        $('resume-panel').classList.add('hidden');
                    }
                }
            };
            // ===== END OF DASHBOARD SECTION =====

            // ===== SETUP SECTION (block setup form: topics, chapters, filters, question count) =====
            const setup = {
                render() {
                    if(!state.bank) return;
                    
                    const topics = new Set();
                    state.bank.questions.forEach(q => { if(q.topic) topics.add(q.topic); });
                    
                    const sel = $('setup-topic');
                    sel.innerHTML = '<option value="all">All Topics</option>';
                    Array.from(topics).sort().forEach(t => {
                        const opt = document.createElement('option');
                        opt.value = t; opt.textContent = t;
                        sel.appendChild(opt);
                    });

                    $('setup-topic').onchange = () => {                         
                        this.renderSections();                         
                        this.updateCount();                     
                    };
                    
                    $$('input[name="q_status"]').forEach(r => r.onchange = () => this.updateCount());

                    this.renderSections();
                    this.updateCount();
                },

                renderSections() {
                    const topic = $('setup-topic').value;
                    const container = $('setup-sections-container');
                    const list = $('setup-sections-list');
                    list.innerHTML = '';

                    if (topic === 'all') {
                        container.classList.add('hidden');
                        return;
                    }

                    const sections = new Set();
                    state.bank.questions.forEach(q => {
                        if (q.topic === topic && q.section) sections.add(q.section);
                    });

                    if (sections.size === 0) {
                        container.classList.add('hidden');
                        return;
                    }

                    container.classList.remove('hidden');
                    
                    Array.from(sections).sort().forEach(sec => {
                        const label = document.createElement('label');
                        label.className = "flex items-center gap-2 cursor-pointer bg-slate-50 dark:bg-slate-800 px-3 py-2 rounded-xl border border-taumd-borderLight dark:border-taumd-borderDark hover:border-taumd-accent transition-colors select-none";
                        
                        label.innerHTML = `
                            <input type="checkbox" value="${data.escapeHTML(sec)}" class="section-checkbox appearance-none w-4 h-4 rounded border-2 border-slate-300 dark:border-slate-600 checked:bg-taumd-accent checked:border-taumd-accent focus:outline-none focus:ring-2 focus:ring-offset-1 focus:ring-taumd-accent/50 transition-colors flex-shrink-0 cursor-pointer checked:bg-[url('data:image/svg+xml;charset=utf-8,%3Csvg%20viewBox%3D%220%200%2016%2016%22%20fill%3D%22%23fff%22%20xmlns%3D%22http%3A%2F%2Fwww.w3.org%2F2000%2Fsvg%22%3E%3Cpath%20d%3D%22M12.207%204.793a1%201%200%20010%201.414l-5%205a1%201%200%2001-1.414%200l-2-2a1%201%200%20011.414-1.414L6.5%209.086l4.293-4.293a1%201%200%20011.414%200z%22%2F%3E%3C%2Fsvg%3E')] bg-center bg-no-repeat bg-[length:100%_100%]">
                            <span class="text-sm font-medium text-taumd-textDark dark:text-slate-200">${data.escapeHTML(sec)}</span>
                        `;
                        
                        label.querySelector('input').addEventListener('change', () => this.updateCount());
                        list.appendChild(label);
                    });
                },

                updateCount() {
                    const filtered = this.getFilteredQuestions();
                    $('setup-match-count').textContent = `${filtered.length} matching questions available.`;
                },

                getFilteredQuestions() {
                    const topic = $('setup-topic').value;
                    const status = document.querySelector('input[name="q_status"]:checked').value;
                    const selectedSections = Array.from(document.querySelectorAll('.section-checkbox:checked')).map(cb => cb.value);
                    
                    return state.bank.questions.filter(q => {
                        if (topic !== 'all' && q.topic !== topic) return false;
                        if (selectedSections.length > 0) {
                            if (!q.section || !selectedSections.includes(q.section)) return false;
                        }
                        const history = state.progress.history[q.id];
                        if (status === 'unanswered' && history) return false;
                        if (status === 'incorrect' && (!history || history.correct)) return false;
                        if (status === 'bookmarked' && !state.progress.bookmarks.includes(q.id)) return false;
                        
                        return true;
                    });
                }
            };
            // ===== END OF SETUP SECTION =====

            // ===== SESSION SECTION (practice block: create, timer, question rendering, answers, navigation, results) =====
            const session = {
                // ----- CREATE, START AND TIMER -----
                create() {
                    const filtered = setup.getFilteredQuestions();
                    if(filtered.length === 0) return ui.showAlert("No Questions", "Your current filters resulted in 0 questions. Please adjust them.");
                    
                    const sizeVal = $('setup-count').value;
                    const count = sizeVal === 'all' ? filtered.length : Math.min(parseInt(sizeVal), filtered.length);
                    
                    const shuffled = [...filtered].sort(() => 0.5 - Math.random());
                    const selectedQs = shuffled.slice(0, count);

                    const mode = $('setup-mode').value;
                    const useTimer = mode === 'exam' && $('setup-timer').checked;

                    state.session = {
                        questions: selectedQs,
                        mode: mode,
                        currentIndex: 0,
                        answers: {},
                        submitted: {}, 
                        flags: {}, 
                        startTime: Date.now(),
                        timerLimit: useTimer ? count * 90 : null, 
                        timeRemaining: useTimer ? count * 90 : null,
                        isFinished: false
                    };

                    this.start();
                },

                start() {
                    if(state.timerInterval) clearInterval(state.timerInterval);
                    
                    if(state.session.timerLimit) {
                        $('ws-timer-panel').classList.remove('hidden');
                        this.updateTimerDisplay();
                        state.timerInterval = setInterval(() => {
                            state.session.timeRemaining--;
                            this.updateTimerDisplay();
                            if(state.session.timeRemaining <= 0) {
                                clearInterval(state.timerInterval);
                                this.finish(true, true); 
                            }
                        }, 1000);
                    } else {
                        $('ws-timer-panel').classList.add('hidden');
                    }

                    this.renderWorkspace();
                    ui.navigate('workspace');
                },

                updateTimerDisplay() {
                    if(!state.session || state.session.timeRemaining === null) return;
                    $('ws-timer-display').textContent = ui.formatTime(state.session.timeRemaining);
                    if(state.session.timeRemaining < 60) $('ws-timer-display').classList.add('text-feedback-danger');
                    else $('ws-timer-display').classList.remove('text-feedback-danger');
                },

                // ----- RENDER THE QUESTION SCREEN -----
                renderWorkspace() {
                    if(!state.session) return;
                    const qIndex = state.session.currentIndex;
                    const q = state.session.questions[qIndex];
                    const isAnswered = !!state.session.answers[qIndex];
                    const isSubmitted = state.session.submitted[qIndex];
                    const isExam = state.session.mode === 'exam';

                    $('ws-counter').textContent = `Q ${qIndex + 1} of ${state.session.questions.length}`;
                    
                    const titleStr = q.section ? `${q.topic} - ${q.section}` : (q.topic || 'General');
                    $('ws-topic').textContent = titleStr;
                    
                    const btnFlag = $('btn-flag');
                    if(state.session.flags[qIndex]) btnFlag.classList.add('text-feedback-warning', 'bg-feedback-warningBg');
                    else btnFlag.classList.remove('text-feedback-warning', 'bg-feedback-warningBg');

                    const btnBook = $('btn-bookmark');
                    if(state.progress.bookmarks.includes(q.id)) btnBook.classList.add('text-taumd-accent', 'bg-teal-50', 'dark:bg-teal-900/20');
                    else btnBook.classList.remove('text-taumd-accent', 'bg-teal-50', 'dark:bg-teal-900/20');

                    $('ws-stem').innerHTML = data.formatContent(q.question);
                    
                    if(q.imagePath) {
                        $('ws-image').src = data.escapeHTML(q.imagePath);$('ws-image-container').classList.remove('hidden');
                    } else {
                        $('ws-image-container').classList.add('hidden');
                    }

                    $('ws-notes').value = state.progress.notes[q.id] || "";
                    $('note-saved-indicator').style.opacity = '0';

                    const optsContainer = $('ws-options');
                    optsContainer.innerHTML = '';
                    
                    for(const [key, text] of Object.entries(q.options)) {
                        const btn = document.createElement('button');
                        btn.className = `option-card text-left p-4 sm:p-5 rounded-2xl flex items-start gap-5 focus:outline-none keyboard-focus group`;
                        btn.dataset.key = key;
                        btn.tabIndex = 0;
                        
                        const marker = document.createElement('div');
                        marker.className = `w-10 h-10 flex-shrink-0 flex items-center justify-center rounded-full border-2 border-slate-200 dark:border-slate-700 font-bold bg-slate-50 dark:bg-slate-800 text-slate-500 dark:text-slate-400 transition-colors group-hover:border-taumd-accent group-hover:text-taumd-accent`;
                        marker.textContent = key;
                        
                        const label = document.createElement('div');
                        label.className = `mt-1.5 flex-grow text-base dark:text-slate-200`;
                        label.innerHTML = data.formatContent(text);

                        btn.appendChild(marker);
                        btn.appendChild(label);

                        if(state.session.answers[qIndex] === key) {
                            btn.classList.add('selected');
                            marker.classList.add('bg-taumd-accent', 'text-white', 'border-taumd-accent');
                            marker.classList.remove('bg-slate-50', 'dark:bg-slate-800', 'text-slate-500', 'dark:text-slate-400');
                        }

                        if(isExam) {
                            btn.onclick = () => this.selectOption(key);
                            btn.onkeydown = (e) => { if(e.key === 'Enter' || e.key === ' ') { e.preventDefault(); this.selectOption(key); } };
                        } else {
                            if(isSubmitted) {
                                btn.classList.add('disabled');
                                if(key === q.correctAnswer) {
                                    btn.classList.add('correct');
                                    marker.className = `w-10 h-10 flex-shrink-0 flex items-center justify-center rounded-full border-2 font-bold bg-feedback-success text-white border-feedback-success`;
                                    marker.textContent = key;
                                } else if (state.session.answers[qIndex] === key) {
                                    btn.classList.add('incorrect');
                                    marker.className = `w-10 h-10 flex-shrink-0 flex items-center justify-center rounded-full border-2 font-bold bg-feedback-danger text-white border-feedback-danger`;
                                    marker.textContent = key;
                                }
                            } else {
                                btn.onclick = () => this.selectOption(key);
                                btn.onkeydown = (e) => { if(e.key === 'Enter' || e.key === ' ') { e.preventDefault(); this.selectOption(key); } };
                            }
                        }
                        optsContainer.appendChild(btn);
                    }

                    if((isExam && isAnswered) || (!isExam && isSubmitted)) {
                        $('ws-confidence-panel').classList.remove('hidden');$$('.conf-btn').forEach(b => b.classList.remove('bg-slate-800', 'text-white', 'dark:bg-slate-700', 'border-transparent'));
                        
                        const confData = state.progress.history[q.id]?.conf;
                        if(confData) {
                            const activeBtn = document.querySelector(`.conf-btn[data-val="${confData}"]`);
                            if(activeBtn) activeBtn.classList.add('bg-slate-800', 'text-white', 'dark:bg-slate-700', 'border-transparent');
                        }
                    } else {
                        $('ws-confidence-panel').classList.add('hidden');
                    }

                    $('btn-ws-prev').disabled = qIndex === 0;
                    $('btn-ws-submit').classList.add('hidden');$('btn-ws-next').classList.add('hidden');
                    $('btn-ws-finish').classList.add('hidden');$('ws-feedback').classList.add('hidden');

                    if(isExam) {
                        if(qIndex < state.session.questions.length - 1) $('btn-ws-next').classList.remove('hidden');
                        else $('btn-ws-finish').classList.remove('hidden');
                    } else {
                        if(isSubmitted) {
                            this.renderFeedback(q, state.session.answers[qIndex]);
                            if(qIndex < state.session.questions.length - 1) $('btn-ws-next').classList.remove('hidden');
                            else $('btn-ws-finish').classList.remove('hidden');
                        } else {
                            $('btn-ws-submit').classList.remove('hidden');
                            if(!isAnswered) $('btn-ws-submit').disabled = true;
                            else $('btn-ws-submit').disabled = false;
                        }
                    }

                    this.renderNavGrid();
                    $('ws-scroll-area').scrollTop = 0;
                },

                // ----- ANSWER SELECTION, SUBMIT AND FEEDBACK -----
                selectOption(key) {
                    const qIndex = state.session.currentIndex;
                    if(state.session.mode === 'tutor' && state.session.submitted[qIndex]) return;

                    state.session.answers[qIndex] = key;
                    this.renderWorkspace(); 
                },

                submit() {
                    const qIndex = state.session.currentIndex;
                    if(!state.session.answers[qIndex]) return;

                    const q = state.session.questions[qIndex];
                    const isCorrect = state.session.answers[qIndex] === q.correctAnswer;
                    
                    state.session.submitted[qIndex] = true;
                    storage.saveAnswer(q.id, isCorrect);
                    
                    this.renderWorkspace();
                },

                renderFeedback(q, userKey) {
                    const fb = $('ws-feedback');
                    const head = $('fb-header');
                    const text = $('fb-text');
                    
                    fb.classList.remove('hidden');
                    if(userKey === q.correctAnswer) {
                        head.innerHTML = `<div class="p-2 bg-feedback-successBg text-feedback-success rounded-xl"><svg class="w-6 h-6" fill="none" stroke="currentColor" stroke-width="3" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" d="M5 13l4 4L19 7"></path></svg></div> <span class="text-feedback-success tracking-tight">Correct Answer</span>`;
                    } else {
                        head.innerHTML = `<div class="p-2 bg-feedback-dangerBg text-feedback-danger rounded-xl"><svg class="w-6 h-6" fill="none" stroke="currentColor" stroke-width="3" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" d="M6 18L18 6M6 6l12 12"></path></svg></div> <span class="text-feedback-danger tracking-tight">Incorrect Answer</span>`;
                    }

                    let content = '';
                    if(q.explanation) content += `<div class="mb-6">${data.formatContent(q.explanation)}</div>`;
                    if(q.optionExplanations) {
                        content += '<div class="space-y-4 mt-6 border-t border-taumd-borderLight dark:border-taumd-borderDark pt-6">';
                        for(const [k, expl] of Object.entries(q.optionExplanations)) {
                            content += `<div class="flex gap-4"><div class="w-8 h-8 shrink-0 flex items-center justify-center rounded-full bg-slate-200 dark:bg-slate-700 text-sm font-bold text-slate-700 dark:text-slate-300">${k}</div><div class="flex-grow pt-1">${data.formatContent(expl)}</div></div>`;
                        }
                        content += '</div>';
                    }
                    if(!content) content = "No detailed explanation provided.";
                    text.innerHTML = content;

                    if(q.educationalObjective) {
                        $('fb-objective-wrap').classList.remove('hidden');$('fb-objective-text').innerHTML = data.formatContent(q.educationalObjective);
                    } else {
                        $('fb-objective-wrap').classList.add('hidden');
                    }
                },

                // ----- NAVIGATION, FLAGS, BOOKMARKS AND CONFIDENCE -----
                next() {
                    if(state.session.currentIndex < state.session.questions.length - 1) {
                        state.session.currentIndex++;
                        this.renderWorkspace();
                    }
                },

                prev() {
                    if(state.session.currentIndex > 0) {
                        state.session.currentIndex--;
                        this.renderWorkspace();
                    }
                },

                jumpTo(idx) {
                    state.session.currentIndex = idx;
                    this.renderWorkspace();
                    if(window.innerWidth <= 1024) ui.toggleSidebar();
                },

                toggleFlag() {
                    const idx = state.session.currentIndex;
                    state.session.flags[idx] = !state.session.flags[idx];
                    this.renderWorkspace();
                },

                toggleBookmark() {
                    const q = state.session.questions[state.session.currentIndex];
                    storage.toggleBookmark(q.id);
                    this.renderWorkspace();
                },

                setConfidence(val) {
                    const q = state.session.questions[state.session.currentIndex];
                    if(state.progress.history[q.id]) {
                        state.progress.history[q.id].conf = val;
                        state.progress.history[q.id].confAt = new Date().toISOString();
                        storage.saveBankProgress();
                        this.renderWorkspace(); 
                    }
                },

                renderNavGrid() {
                    const grid = $('ws-nav-grid');
                    grid.innerHTML = '';
                    for(let i=0; i<state.session.questions.length; i++) {
                        const btn = document.createElement('button');
                        btn.className = `h-10 w-full rounded-xl text-xs font-bold flex items-center justify-center border-2 transition-all`;
                        btn.textContent = i + 1;
                        
                        let isAnsw = !!state.session.answers[i];
                        let isFlag = !!state.session.flags[i];
                        
                        if (i === state.session.currentIndex) {
                            btn.classList.add('ring-4', 'ring-taumd-accent/30', 'dark:ring-taumd-accent/20');
                        }

                        if (isFlag) {
                            btn.classList.add('bg-feedback-warningBg', 'border-feedback-warning', 'text-feedback-warning', 'dark:bg-yellow-900/30');
                        } else if (isAnsw) {
                            btn.classList.add('bg-taumd-primary', 'border-taumd-primary', 'text-white', 'dark:bg-taumd-accent', 'dark:border-taumd-accent');
                        } else {
                            btn.classList.add('bg-white', 'border-slate-200', 'text-slate-500', 'dark:bg-slate-800', 'dark:border-slate-700', 'dark:text-slate-400');
                        }

                        btn.onclick = () => this.jumpTo(i);
                        grid.appendChild(btn);
                    }
                },

                // ----- FINISH BLOCK AND RESULTS -----
                finish(promptUser = false, autoSubmit = false) {
                    if(promptUser && !autoSubmit) {
                        const unans = state.session.questions.length - Object.keys(state.session.answers).length;
                        let msg = "Are you sure you want to end this block?";
                        if(unans > 0 && state.session.mode === 'exam') msg += ` You have ${unans} unanswered questions.`;
                        
                        ui.showConfirm("End Block", msg, () => this.processResults());
                    } else if (autoSubmit) {
                        ui.showAlert("Time Up", "Your time has expired. The block will now be submitted.");
                        this.processResults();
                    } else {
                        this.processResults();
                    }
                },

                processResults() {
                    if(state.timerInterval) clearInterval(state.timerInterval);
                    state.session.isFinished = true;
                    
                    let correctCount = 0;
                    let wrongCount = 0;
                    let omittedCount = 0;

                    if(state.session.mode === 'exam') {
                        state.session.questions.forEach((q, idx) => {
                            const ans = state.session.answers[idx];
                            if(!ans) {
                                omittedCount++;
                            } else {
                                const isC = ans === q.correctAnswer;
                                if(isC) correctCount++; else wrongCount++;
                                storage.saveAnswer(q.id, isC); 
                            }
                        });
                    } else {
                        state.session.questions.forEach((q, idx) => {
                             if(state.session.submitted[idx]) {
                                 if(state.session.answers[idx] === q.correctAnswer) correctCount++;
                                 else wrongCount++;
                             } else {
                                 omittedCount++;
                             }
                        });
                    }

                    this.renderResults(correctCount, wrongCount, omittedCount);
                },

                renderResults(c, w, o) {
                    const total = state.session.questions.length;
                    const attempted = c + w;
                    const score = attempted > 0 ? Math.round((c / total) * 100) : 0; 

                    $('res-correct').textContent = c;
                    $('res-incorrect').textContent = w;
                    $('res-omitted').textContent = o;
                    $('res-score').textContent = score + '%';
                    
                    const ring = $('res-ring');
                    const offset = 276.46 - (276.46 * score) / 100;
                    ring.style.strokeDashoffset = offset;

                    if(w > 0) $('btn-res-mistakes').classList.remove('hidden');
                    else $('btn-res-mistakes').classList.add('hidden');

                    const rList = $('res-list');
                    rList.innerHTML = '';
                    
                    state.session.questions.forEach((q, idx) => {
                        const ans = state.session.answers[idx];
                        const isSub = state.session.mode === 'exam' || state.session.submitted[idx];
                        const isCorrect = isSub && ans === q.correctAnswer;
                        
                        const item = document.createElement('div');
                        item.className = 'p-5 hover:bg-slate-50 dark:hover:bg-slate-800 transition-colors flex flex-col sm:flex-row justify-between sm:items-center gap-4 cursor-pointer group';
                        
                        let statusIcon = `<span class="px-3 py-1 bg-slate-100 dark:bg-slate-700 text-taumd-textMuted text-xs font-bold rounded-lg uppercase tracking-wider">Omitted</span>`;
                        if(isSub && ans) {
                            if(isCorrect) statusIcon = `<span class="px-3 py-1 bg-feedback-successBg text-feedback-success text-xs font-bold rounded-lg uppercase tracking-wider">Correct</span>`;
                            else statusIcon = `<span class="px-3 py-1 bg-feedback-dangerBg text-feedback-danger text-xs font-bold rounded-lg uppercase tracking-wider">Incorrect</span>`;
                        }

                        item.innerHTML = `
                            <div class="flex items-center gap-4">
                                <span class="text-base font-black w-8 text-slate-300 dark:text-slate-600">Q${idx+1}</span>
                                <div>
                                    <div class="text-sm font-semibold text-taumd-textDark dark:text-slate-200 line-clamp-1 group-hover:text-taumd-accent transition-colors">${data.escapeHTML(q.question).substring(0,80)}...</div>
                                    <div class="text-xs font-medium text-slate-400 mt-1">${q.topic || 'General'} ${q.section ? '• ' + q.section : ''}</div>
                                </div>
                            </div>
                            <div class="flex items-center gap-4 shrink-0">
                                ${statusIcon}
                                <svg class="w-5 h-5 text-slate-300 group-hover:text-taumd-accent transition-colors" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M9 5l7 7-7 7"></path></svg>
                            </div>
                        `;
                        
                        item.onclick = () => {
                            state.session.currentIndex = idx;
                            if(state.session.mode === 'exam' && ans) state.session.submitted[idx] = true;
                            this.renderWorkspace();
                            $('btn-ws-submit').classList.add('hidden'); 
                            ui.navigate('workspace');
                        };
                        
                        rList.appendChild(item);
                    });

                    ui.navigate('results');
                }
            };
            // ===== END OF SESSION SECTION =====

            // ===== EXPLORER SECTION (topics overview, bookmarks list, search) =====
            const explorer = {
                render() {
                    if(!state.bank) return;
                    this.renderTopics();
                    this.renderBookmarks();
                },
                renderTopics() {
                    const cont = $('tab-topics');
                    cont.innerHTML = '';
                    
                    const topicData = {};
                    state.bank.questions.forEach(q => {
                        const t = q.topic || 'Uncategorized';
                        if(!topicData[t]) topicData[t] = { total: 0, attempted: 0, correct: 0 };
                        topicData[t].total++;
                        
                        const h = state.progress.history[q.id];
                        if(h) {
                            topicData[t].attempted++;
                            if(h.correct) topicData[t].correct++;
                        }
                    });

                    for(const [t, stats] of Object.entries(topicData).sort()) {
                        const acc = stats.attempted > 0 ? Math.round((stats.correct/stats.attempted)*100) : 0;
                        const row = document.createElement('div');
                        row.className = 'p-6 flex flex-col sm:flex-row justify-between sm:items-center gap-6 hover:bg-slate-50 dark:hover:bg-slate-800 transition-colors group';
                        row.innerHTML = `
                            <div>
                                <h4 class="font-bold text-lg text-taumd-primary dark:text-white mb-1 group-hover:text-taumd-accent transition-colors">${data.escapeHTML(t)}</h4>
                                <div class="text-sm font-medium text-slate-500 flex gap-4">
                                    <span class="flex items-center gap-1.5"><svg class="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M19 11H5m14 0a2 2 0 012 2v6a2 2 0 01-2 2H5a2 2 0 01-2-2v-6a2 2 0 012-2m14 0V9a2 2 0 00-2-2M5 11V9a2 2 0 002-2m0 0V5a2 2 0 012-2h6a2 2 0 012 2v2M7 7h10"></path></svg> ${stats.total} Questions</span>
                                    <span class="flex items-center gap-1.5"><svg class="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M11 5H6a2 2 0 00-2 2v11a2 2 0 002 2h11a2 2 0 002-2v-5m-1.414-9.414a2 2 0 112.828 2.828L11.828 15H9v-2.828l8.586-8.586z"></path></svg> ${stats.attempted} Attempted</span>
                                </div>
                            </div>
                            <div class="flex items-center gap-6 shrink-0">
                                <div class="flex items-center gap-3 w-40">
                                    <div class="flex-grow bg-slate-100 dark:bg-slate-700 rounded-full h-2">
                                        <div class="bg-taumd-accent h-2 rounded-full" style="width: ${acc}%"></div>
                                    </div>
                                    <span class="text-sm font-black w-10 text-right text-slate-700 dark:text-slate-300">${acc}%</span>
                                </div>
                                <button onclick="TAUMD.startReviewMistakes('${t}')" class="px-5 py-2 text-sm font-bold border-2 border-taumd-primary text-taumd-primary dark:border-taumd-accent dark:text-taumd-accent rounded-lg hover:bg-taumd-primary hover:text-white dark:hover:bg-taumd-accent dark:hover:text-white transition-all shadow-sm">Study</button>
                            </div>
                        `;
                        cont.appendChild(row);
                    }
                },
                renderBookmarks() {
                    const cont = $('tab-bookmarks');
                    cont.innerHTML = '';
                    if(state.progress.bookmarks.length === 0) {
                        cont.innerHTML = '<div class="p-12 text-center text-taumd-textMuted text-lg font-medium">You have no bookmarked questions.</div>';
                        return;
                    }

                    state.progress.bookmarks.forEach(id => {
                        const q = state.bank.questions.find(x => x.id === id);
                        if(!q) return;
                        const row = document.createElement('div');
                        row.className = 'p-6 hover:bg-slate-50 dark:hover:bg-slate-800 flex justify-between items-center cursor-pointer group transition-colors';
                        row.innerHTML = `
                            <div class="pr-6">
                                <div class="text-base font-semibold text-taumd-textDark dark:text-slate-200 line-clamp-2 mb-2 group-hover:text-taumd-accent transition-colors">${data.escapeHTML(q.question).substring(0,120)}...</div>
                                <div class="text-xs font-bold tracking-wide uppercase text-slate-400 flex items-center gap-2">
                                    <span class="bg-slate-100 dark:bg-slate-700 px-2 py-1 rounded text-slate-700 dark:text-slate-300">${q.id}</span>
                                    <span>${data.escapeHTML(q.topic)} ${q.section ? '• ' + data.escapeHTML(q.section) : ''}</span>
                                </div>
                            </div>
                            <button onclick="event.stopPropagation(); TAUMD.storage.toggleBookmark('${q.id}'); TAUMD.explorer.renderBookmarks();" class="p-3 text-taumd-accent bg-teal-50 dark:bg-teal-900/20 hover:bg-rose-50 hover:text-feedback-danger rounded-xl transition-all" title="Remove Bookmark">
                                <svg class="w-5 h-5 fill-current" viewBox="0 0 24 24"><path d="M5 5a2 2 0 012-2h10a2 2 0 012 2v16l-7-3.5L5 21V5z"></path></svg>
                            </button>
                        `;
                        row.onclick = () => this.launchSingleQuestion(q);
                        cont.appendChild(row);
                    });
                },
                doSearch(query) {
                    if(!query.trim()) {
                        $('tab-search-results').innerHTML = '<div class="p-12 text-center text-taumd-textMuted text-lg font-medium">Type in the search bar above to find questions.</div>';
                        return;
                    }
                    const term = query.toLowerCase();
                    const results = state.bank.questions.filter(q => 
                        (q.id && q.id.toLowerCase().includes(term)) ||
                        (q.question && q.question.toLowerCase().includes(term)) ||
                        (q.topic && q.topic.toLowerCase().includes(term)) ||
                        (q.section && q.section.toLowerCase().includes(term))
                    ).slice(0, 50);

                    const cont = $('tab-search-results');
                    cont.innerHTML = '';
                    
                    if(results.length === 0) {
                        cont.innerHTML = '<div class="p-12 text-center text-taumd-textMuted text-lg font-medium">No questions found matching your query.</div>';
                        return;
                    }

                    results.forEach(q => {
                        const row = document.createElement('div');
                        row.className = 'p-5 hover:bg-slate-50 dark:hover:bg-slate-800 cursor-pointer transition-colors group';
                        row.innerHTML = `
                            <div class="text-base font-semibold text-taumd-textDark dark:text-slate-200 line-clamp-2 mb-2 group-hover:text-taumd-accent transition-colors">${data.escapeHTML(q.question)}</div>
                            <div class="text-xs font-bold tracking-wide uppercase text-slate-400 flex items-center gap-2">
                                <span class="bg-slate-100 dark:bg-slate-700 px-2 py-1 rounded text-slate-600 dark:text-slate-300">${q.id}</span>
                                <span>${data.escapeHTML(q.topic || 'General')} ${q.section ? '• ' + data.escapeHTML(q.section) : ''}</span>
                            </div>
                        `;
                        row.onclick = () => this.launchSingleQuestion(q);
                        cont.appendChild(row);
                    });
                },
                launchSingleQuestion(q) {
                    state.session = {
                        questions: [q], mode: 'tutor', currentIndex: 0,
                        answers: {}, submitted: {}, flags: {}, isFinished: false, timeRemaining: null
                    };
                    session.start();
                }
            };
            // ===== END OF EXPLORER SECTION =====

            // ===== PUBLIC API AND STARTUP SECTION (what the app exposes: init and review mistakes) =====
            return {
                ui, storage, data, session, explorer, account, settings: storage,
                async init() {
                    ui.init();
                    storage.loadSettings();
                    ui.navigate('loading');
                    await account.init();     // restores a previous Google sign-in (if any) before the bank loads
                    data.initLoad();
                    
                    document.addEventListener('keydown', (e) => {
                        if(!state.session || state.session.isFinished) return;
                        if(e.target.tagName === 'INPUT' || e.target.tagName === 'TEXTAREA') return; 
                        
                        if(e.key === 'ArrowRight') session.next();
                        else if (e.key === 'ArrowLeft') session.prev();
                        else if (['1','2','3','4','5','6','7','8','9'].includes(e.key)) {
                            const index = parseInt(e.key) - 1;
                            const optionsMap = Object.keys(state.session.questions[state.session.currentIndex].options);
                            if(index < optionsMap.length) session.selectOption(optionsMap[index]);
                        }
                    });
                },
                startReviewMistakes(topicFilter = null) {
                    const incorrectQs = state.bank.questions.filter(q => {
                        const h = state.progress.history[q.id];
                        if(!h || h.correct) return false;
                        if(topicFilter && q.topic !== topicFilter) return false;
                        return true;
                    });
                    
                    if(incorrectQs.length === 0) return ui.showAlert("All Good", "No mistakes to review in this category.");
                    
                    ui.navigate('setup');
                    $('setup-topic').value = topicFilter ? topicFilter : 'all';
                    
                    if (topicFilter) setup.renderSections();

                    document.querySelector('input[name="q_status"][value="incorrect"]').checked = true;
                    setup.updateCount();
                }
            };
            // ===== END OF PUBLIC API AND STARTUP SECTION =====
        })();
        // ===== END OF APP MODULE SECTION =====

        // ===== GLOBAL EXPORT AND STARTUP SECTION (expose TAUMD to the inline onclick handlers and start on page load) =====
        // Make TAUMD available globally to keep existing inline onclick handlers working
        window.TAUMD = TAUMD;

        document.addEventListener('DOMContentLoaded', () => { TAUMD.init(); });
        // ===== END OF GLOBAL EXPORT AND STARTUP SECTION =====
    </script>
